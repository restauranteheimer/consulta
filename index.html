// =================================================================== // WEBHOOK COMPLETO - Com suporte a CORS // =================================================================== function doPost(e) { // Configura CORS para permitir requisições de qualquer origem const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Content-Type': 'application/json' }; try { const params = JSON.parse(e.postData.contents); const intentName = params.queryResult.intent.displayName; let resposta; if (intentName === 'ler_registro') { resposta = buscarRapido(params); } else if (intentName === 'RegistrarCliente' || intentName === 'cadastrardireto') { resposta = salvarRapido(params); } else { resposta = { fulfillmentText: "Intent não reconhecida: " + intentName }; } return ContentService .createTextOutput(JSON.stringify(resposta)) .setMimeType(ContentService.MimeType.JSON) .setHeaders(corsHeaders); } catch (err) { console.error("Erro:", err); return ContentService .createTextOutput(JSON.stringify({ fulfillmentText: "Erro interno: " + err.toString() })) .setMimeType(ContentService.MimeType.JSON) .setHeaders(corsHeaders); } } // Função para OPTIONS (preflight CORS) function doOptions(e) { return ContentService .createTextOutput('') .setMimeType(ContentService.MimeType.TEXT) .setHeaders({ 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Access-Control-Max-Age': '86400' }); } // ==================== EXTRAIR NOME ==================== function extrairNome(params) { const p = params.queryResult.parameters['person'] || params.queryResult.parameters['sysperson']; if (!p) return 'Cliente'; if (typeof p === 'string') return p.trim(); if (p.name) return p.name.trim(); if (Array.isArray(p) && p[0] && p[0].name) return p[0].name.trim(); if (Array.isArray(p) && typeof p[0] === 'string') return p.join(' ').trim(); return 'Cliente'; } // ==================== EXTRAIR TELEFONE ==================== function extrairTelefone(raw) { if (!raw) return null; let nums = raw.toString().replace(/\D/g, ''); if (nums.startsWith('55')) nums = nums.substring(2); if (nums.length > 11) { for (let i = 0; i <= nums.length - 11; i++) { const ddd = parseInt(nums.substring(i, i+2)); if (ddd >= 11 && ddd <= 99) { nums = nums.substring(i, i+11); break; } } } if (nums.length !== 10 && nums.length !== 11) return null; const ddd = nums.substring(0, 2); let numero = nums.substring(2); if (numero.length === 8) numero = '9' + numero; const completo = '55' + ddd + numero; return { completo: completo, link: 'https://wa.me/' + completo }; } // ==================== SALVAR ==================== function salvarRapido(params) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const nome = extrairNome(params); let telefoneRaw = params.queryResult.parameters['phone-number'] || ''; if (Array.isArray(telefoneRaw)) telefoneRaw = telefoneRaw[0]; if (telefoneRaw && typeof telefoneRaw === 'object') telefoneRaw = telefoneRaw.amount || ''; const telefone = extrairTelefone(telefoneRaw); if (!telefone) { return { fulfillmentText: "❌ Telefone inválido! Use 81 99999-9999" }; } const data = sheet.getDataRange().getValues(); let linhaExistente = -1; for (let i = 1; i < data.length; i++) { if (data[i][1] === telefone.completo) { linhaExistente = i; break; } } if (linhaExistente !== -1) { const linha = linhaExistente + 1; sheet.getRange(linha, 1).setValue(nome); sheet.getRange(linha, 2).setValue(telefone.completo); sheet.getRange(linha, 3).setValue(telefone.link); sheet.getRange(linha, 6).setValue(new Date().toLocaleString('pt-BR')); return { fulfillmentText: `🔄 ${nome} atualizado!\n${telefone.link}` }; } else { sheet.appendRow([nome, telefone.completo, telefone.link, '', '', new Date().toLocaleString('pt-BR')]); return { fulfillmentText: `✅ ${nome} cadastrado!\n${telefone.link}` }; } } // ==================== BUSCAR (MÁXIMO 3 RESULTADOS) ==================== function buscarRapido(params) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const nomeBusca = extrairNome(params); if (!nomeBusca || nomeBusca === 'Cliente') { return { fulfillmentText: "Qual nome você quer buscar?" }; } const busca = nomeBusca.toLowerCase().trim(); const data = sheet.getDataRange().getValues(); const resultados = []; for (let i = 1; i < data.length && resultados.length < 4; i++) { const nomeCelula = data[i][0]; if (nomeCelula && nomeCelula.toString().toLowerCase().includes(busca)) { resultados.push({ nome: data[i][0], whats: data[i][2] || data[i][1] }); } } if (resultados.length > 3) { return { fulfillmentText: `❌ Encontrei mais de 3 clientes com nome parecido com "${nomeBusca}". Por favor, seja mais específico.` }; } if (resultados.length === 0) { return { fulfillmentText: `❌ Não encontrei nenhum cliente com nome parecido com "${nomeBusca}".` }; } let mensagem = `🔍 Encontrei ${resultados.length} cliente(s):\n\n`; for (let i = 0; i < resultados.length; i++) { const r = resultados[i]; mensagem += `${i+1}. *${r.nome}*\n 📞 ${r.whats}\n\n`; } return { fulfillmentText: mensagem }; }
